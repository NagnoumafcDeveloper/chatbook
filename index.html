document.addEventListener("DOMContentLoaded", () => { // 1. Gestion des J'aime const likeElements = document.querySelectorAll('.like-btn, [id*="like"], .fa-heart'); likeElements.forEach(el => { el.style.cursor = 'pointer'; el.addEventListener('click', () => { let countSpan = el.querySelector('span') || el.nextElementSibling; if (countSpan && countSpan.innerText.includes('K')) { let val = parseFloat(countSpan.innerText) + 0.1; countSpan.innerText = val.toFixed(1) + 'K'; el.style.color = '#ff4d4d'; } }); }); // 2. Gestion des Commentaires (envoi et ajout) const sendButtons = document.querySelectorAll('.send-btn, button'); sendButtons.forEach(btn => { if (btn.innerText.toLowerCase().includes('envoyer') || btn.id.includes('send')) { btn.addEventListener('click', () => { let input = document.querySelector('.comment-input, input[type="text"]'); let list = document.querySelector('.comment-list, .comments'); if (input && list && input.value.trim() !== '') { let newC = document.createElement('div'); newC.className = 'comment'; newC.innerHTML = `Vous: ${input.value}`; list.appendChild(newC); input.value = ''; } }); } }); // 3. Gestion du Partage const shareElements = document.querySelectorAll('.share-btn, [id*="share"]'); shareElements.forEach(el => { el.style.cursor = 'pointer'; el.addEventListener('click', () => { navigator.clipboard.writeText(window.location.href); alert('Lien copié !'); }); }); });
